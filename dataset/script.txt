DROP DATABASE IF EXISTS mudancas;
CREATE DATABASE IF NOT EXISTS mudancas CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE mudancas;

-- 1. CRIAÇÃO DAS TABELAS
CREATE TABLE empresa (
    id_empresa INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(120) NOT NULL UNIQUE,
    endereco VARCHAR(200) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE empresa_telefone (
    id_empresa INT NOT NULL,
    telefone VARCHAR(20) NOT NULL,
    PRIMARY KEY (id_empresa, telefone),
    FOREIGN KEY (id_empresa) REFERENCES empresa(id_empresa) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE cidade (
    id_cidade INT AUTO_INCREMENT PRIMARY KEY,
    nome_cidade VARCHAR(100) NOT NULL,
    estado CHAR(2) NOT NULL,
    UNIQUE (nome_cidade, estado)
) ENGINE=InnoDB;

CREATE TABLE servico (
    nome_servico VARCHAR(80) PRIMARY KEY,
    tipo_especializacao VARCHAR(12) NOT NULL DEFAULT 'NENHUM'
        CHECK (tipo_especializacao IN ('GUINDASTE', 'TRANSPORTE', 'NENHUM'))
) ENGINE=InnoDB;

CREATE TABLE guindaste (
    nome_servico VARCHAR(80) PRIMARY KEY,
    tamanho_base VARCHAR(30) NOT NULL,
    altura DECIMAL(6,2) NOT NULL,
    bonus_aumento DECIMAL(5,2) NOT NULL,
    FOREIGN KEY (nome_servico) REFERENCES servico(nome_servico) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE transporte (
    nome_servico VARCHAR(80) PRIMARY KEY,
    FOREIGN KEY (nome_servico) REFERENCES servico(nome_servico) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE transporte_acrescimo (
    nome_servico VARCHAR(80) NOT NULL,
    peso_limite DECIMAL(10,2) NOT NULL,
    percentual_acrescimo DECIMAL(5,2) NOT NULL,
    PRIMARY KEY (nome_servico, peso_limite),
    FOREIGN KEY (nome_servico) REFERENCES transporte(nome_servico) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE oferecem (
    id_empresa INT NOT NULL,
    id_cidade INT NOT NULL,
    nome_servico VARCHAR(80) NOT NULL,
    preco_hora DECIMAL(10,2) NOT NULL CHECK (preco_hora > 0),
    PRIMARY KEY (id_empresa, id_cidade, nome_servico),
    FOREIGN KEY (id_empresa) REFERENCES empresa(id_empresa) ON DELETE CASCADE,
    FOREIGN KEY (id_cidade) REFERENCES cidade(id_cidade) ON DELETE CASCADE,
    FOREIGN KEY (nome_servico) REFERENCES servico(nome_servico) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE cliente (
    codigo_cliente INT AUTO_INCREMENT PRIMARY KEY,
    cpf CHAR(11) NOT NULL UNIQUE,
    rg VARCHAR(20) NOT NULL UNIQUE,
    nome_completo VARCHAR(150) NOT NULL,
    endereco VARCHAR(200) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE cliente_telefone (
    codigo_cliente INT NOT NULL,
    telefone VARCHAR(20) NOT NULL,
    PRIMARY KEY (codigo_cliente, telefone),
    FOREIGN KEY (codigo_cliente) REFERENCES cliente(codigo_cliente) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE funcionario (
    cpf_func CHAR(11) PRIMARY KEY,
    rg_func VARCHAR(20) NOT NULL UNIQUE,
    nome_completo VARCHAR(150) NOT NULL,
    endereco VARCHAR(200) NOT NULL,
    tipo_func VARCHAR(30) NOT NULL
        CHECK (tipo_func IN ('MOTORISTA', 'OPERADOR_GUINCHO', 'GERENTE_MUDANCAS', 'AJUDANTE')),
    salario DECIMAL(10,2) NOT NULL CHECK (salario > 0)
) ENGINE=InnoDB;

CREATE TABLE funcionario_telefone (
    cpf_func CHAR(11) NOT NULL,
    telefone VARCHAR(20) NOT NULL,
    PRIMARY KEY (cpf_func, telefone),
    FOREIGN KEY (cpf_func) REFERENCES funcionario(cpf_func) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE funcionario_empresa (
    cpf_func CHAR(11) NOT NULL,
    id_empresa INT NOT NULL,
    data_inicio DATE NOT NULL,
    data_fim DATE,
    telefone_empresa VARCHAR(20),
    PRIMARY KEY (cpf_func, id_empresa, data_inicio),
    FOREIGN KEY (cpf_func) REFERENCES funcionario(cpf_func) ON DELETE CASCADE,
    FOREIGN KEY (id_empresa) REFERENCES empresa(id_empresa) ON DELETE CASCADE,
    CHECK (data_fim IS NULL OR data_fim >= data_inicio)
) ENGINE=InnoDB;

CREATE TABLE pedido (
    codigo_pedido INT AUTO_INCREMENT PRIMARY KEY,
    id_empresa INT NOT NULL,
    codigo_cliente INT NOT NULL,
    id_cidade_origem INT,
    endereco_origem VARCHAR(200),
    id_cidade_destino INT,
    endereco_destino VARCHAR(200),
    data_solicitacao DATE NOT NULL,
    data_resolucao DATE,
    situacao VARCHAR(10) NOT NULL DEFAULT 'PENDENTE'
        CHECK (situacao IN ('PENDENTE', 'ACEITA', 'REJEITADA')),
    preco_total DECIMAL(12,2) NOT NULL DEFAULT 0,
    FOREIGN KEY (id_empresa) REFERENCES empresa(id_empresa),
    FOREIGN KEY (codigo_cliente) REFERENCES cliente(codigo_cliente),
    FOREIGN KEY (id_cidade_origem) REFERENCES cidade(id_cidade),
    FOREIGN KEY (id_cidade_destino) REFERENCES cidade(id_cidade),
    CHECK (data_resolucao IS NULL OR data_resolucao >= data_solicitacao)
) ENGINE=InnoDB;

CREATE TABLE pedido_servico (
    codigo_pedido INT NOT NULL,
    nome_servico VARCHAR(80) NOT NULL,
    tempo_duracao DECIMAL(6,2) NOT NULL CHECK (tempo_duracao > 0),
    data_fim DATE,
    peso_carga DECIMAL(10,2) CHECK (peso_carga IS NULL OR peso_carga > 0),
    preco DECIMAL(12,2) NOT NULL DEFAULT 0,
    PRIMARY KEY (codigo_pedido, nome_servico),
    FOREIGN KEY (codigo_pedido) REFERENCES pedido(codigo_pedido) ON DELETE CASCADE,
    FOREIGN KEY (nome_servico) REFERENCES servico(nome_servico)
) ENGINE=InnoDB;

CREATE TABLE pedido_servico_funcionario (
    codigo_pedido INT NOT NULL,
    nome_servico VARCHAR(80) NOT NULL,
    cpf_func CHAR(11) NOT NULL,
    PRIMARY KEY (codigo_pedido, nome_servico, cpf_func),
    FOREIGN KEY (cpf_func) REFERENCES funcionario(cpf_func),
    FOREIGN KEY (codigo_pedido, nome_servico)
        REFERENCES pedido_servico(codigo_pedido, nome_servico) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE INDEX idx_pedido_empresa ON pedido(id_empresa);
CREATE INDEX idx_pedido_cliente ON pedido(codigo_cliente);
CREATE INDEX idx_pedido_cidade_origem ON pedido(id_cidade_origem);
CREATE INDEX idx_pedido_servico_servico ON pedido_servico(nome_servico);
CREATE INDEX idx_oferecem_cidade ON oferecem(id_cidade);

DELIMITER $$

CREATE TRIGGER trg_marca_guindaste
AFTER INSERT ON guindaste
FOR EACH ROW
BEGIN
    DECLARE v_tipo_atual VARCHAR(12);
    SELECT tipo_especializacao INTO v_tipo_atual FROM servico WHERE nome_servico = NEW.nome_servico;
    IF v_tipo_atual = 'TRANSPORTE' THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Servico ja e do tipo TRANSPORTE; hierarquia e disjunta';
    END IF;
    UPDATE servico SET tipo_especializacao = 'GUINDASTE' WHERE nome_servico = NEW.nome_servico;
END$$

CREATE TRIGGER trg_marca_transporte
AFTER INSERT ON transporte
FOR EACH ROW
BEGIN

        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Servico ja e do tipo GUINDASTE; hierarquia e disjunta';
    END IF;
    UPDATE servico SET tipo_especializacao = 'TRANSPORTE' WHERE nome_servico = NEW.nome_servico;
END$$

CREATE TRIGGER trg_calcula_preco_ins
BEFORE INSERT ON pedido_servico
FOR EACH ROW
BEGIN
    DECLARE v_id_empresa INT;
    DECLARE v_id_cidade INT;
    DECLARE v_preco_hora DECIMAL(10,2);
    DECLARE v_tipo VARCHAR(12);
    DECLARE v_bonus DECIMAL(5,2);
    DECLARE v_percentual DECIMAL(5,2);
    DECLARE v_base DECIMAL(12,4);

    SELECT id_empresa, COALESCE(id_cidade_origem, id_cidade_destino)
      INTO v_id_empresa, v_id_cidade
      FROM pedido
      WHERE codigo_pedido = NEW.codigo_pedido;

    IF v_id_cidade IS NULL THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Pedido sem cidade de origem/destino definida: impossivel calcular preco';
    END IF;

    SELECT preco_hora INTO v_preco_hora
      FROM oferecem
      WHERE id_empresa = v_id_empresa
        AND id_cidade = v_id_cidade
        AND nome_servico = NEW.nome_servico;

    IF v_preco_hora IS NULL THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'A empresa nao oferece este servico na cidade informada no pedido';
    END IF;

    SET v_base = v_preco_hora * NEW.tempo_duracao;

    SELECT tipo_especializacao INTO v_tipo FROM servico WHERE nome_servico = NEW.nome_servico;

    IF v_tipo = 'TRANSPORTE' AND NEW.peso_carga IS NOT NULL THEN
        SELECT percentual_acrescimo INTO v_percentual
          FROM transporte_acrescimo
          WHERE nome_servico = NEW.nome_servico
            AND peso_limite <= NEW.peso_carga
          ORDER BY peso_limite DESC
          LIMIT 1;
        IF v_percentual IS NOT NULL THEN
            SET v_base = v_base * (1 + v_percentual / 100.0);
        END IF;
    ELSEIF v_tipo = 'GUINDASTE' THEN
        SELECT bonus_aumento INTO v_bonus FROM guindaste WHERE nome_servico = NEW.nome_servico;
        IF v_bonus IS NOT NULL THEN
            SET v_base = v_base * (1 + v_bonus / 100.0);
        END IF;
    END IF;

    SET NEW.preco = ROUND(v_base, 2);
END$$

CREATE TRIGGER trg_atualiza_total_ins
AFTER INSERT ON pedido_servico
FOR EACH ROW
BEGIN
    UPDATE pedido
       SET preco_total = COALESCE(
             (SELECT SUM(preco) FROM pedido_servico WHERE codigo_pedido = NEW.codigo_pedido), 0)
     WHERE codigo_pedido = NEW.codigo_pedido;
END$$

CREATE TRIGGER trg_valida_cidade_ins
BEFORE INSERT ON pedido
FOR EACH ROW

            SET MESSAGE_TEXT = 'A empresa nao presta servico na cidade de origem informada';

        SELECT 1 FROM oferecem WHERE id_empresa = NEW.id_empresa AND id_cidade = NEW.id_cidade_destino
    ) THEN
        SIGNAL SQLSTATE '45000'
